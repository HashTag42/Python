# isPrime(number)
	import math
	def isPrime(number):
	    isPrime = True
	    if number < 2:
	        isPrime = False
	    elif number == 2:
	        isPrime = True
	    elif number % 2 == 0:
	        isPrime = False
	    else:
	        stop = int(math.sqrt(number)) + 1
	        for d in range (3, stop, 2):
	            if ((number % d) == 0):
	                isPrime = False
	                break
	    return isPrime

# Factorial (iterative)
	def factorial_iterative(n):
	    factorial = 1
	    for i in range(1, n + 1):
	        factorial *= i
	    return factorial

# Factorial (recursive)
	def factorial_recursive(n):
	    # Base case: if n is 0 or 1, return 1
	    if n == 0 or n == 1:
	        return 1
	    # Recursive case: n! = n * (n-1)!
	    else:
	        return n * factorial_recursive(n - 1)

# Fibonacci (iterative)
	def fibonacci(n):
	    if n <= 0:
	        return 0
	    elif n == 1:
	        return 1
	    a, b = 0, 1
	    for _ in range(2, n + 1):
	        a, b = b, a + b
	    return b

# Fibonacci (recursive)
	def fibonacci(n):
	    if n <= 0:
	        return 0
	    elif n == 1:
	        return 1
	    else:
	        return fibonacci(n-1) + fibonacci(n-2)

# Height of a Binary Tree (recursive)
	def height(root):
	    h = 0
	    if root is None:
	        h = -1
	    else:
	        lHeight = height(root.left)
	        rHeight = height(root.right)
	        h += 1 + max(lHeight, rHeight)
	    return h
